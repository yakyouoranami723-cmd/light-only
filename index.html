async function callClientProvider(p,payload){ const type=String(p.type||'').toLowerCase(); const system=String(payload.system||''); const messages=Array.isArray(payload.messages)?payload.messages.slice(-20):[]; const maxTokens=Math.max(100,Math.min(2200,Number(payload.maxOutputTokens||800))); const jsonMode=Boolean(payload.jsonMode); let url=providerUrl(p); let headers={'Content-Type':'application/json'}; let body={}; if(!url){ throw new Error('AIプロバイダーの設定が不正です。'); } if(type==='gemini'){ headers['x-goog-api-key']=p.apiKey; const contents=messages.map(m=>({ role:m.role==='assistant'?'model':'user', parts:[ { text:String(m.text||'').slice(0,14000) }, ...((m.images||[]).slice(0,2).map(img=>({ inline_data:{ mime_type:String(img.mimeType||'image/jpeg'), data:String(img.base64||'') } }))) ] })); body={ contents, systemInstruction:{ parts:[ { text:system.slice(0,16000) } ] }, generationConfig:{ maxOutputTokens:maxTokens } }; if(jsonMode){ body.generationConfig.responseMimeType='application/json'; } }else if(type==='openai'||type==='openai_compatible'){ headers.Authorization=`Bearer ${p.apiKey}`; const out=[]; if(system){ out.push({ role:'system', content:system.slice(0,16000) }); } for(const m of messages){ const role=m.role==='assistant'?'assistant':'user'; if(m.images?.length){ const content=[ { type:'text', text:String(m.text||'').slice(0,14000) } ]; for(const img of m.images.slice(0,2)){ content.push({ type:'image_url', image_url:{ url:`data:${String(img.mimeType||'image/jpeg')};base64,${String(img.base64||'')}` } }); } out.push({ role, content }); }else{ out.push({ role, content:String(m.text||'').slice(0,14000) }); } } body={ model:p.model, messages:out, max_tokens:maxTokens }; if(jsonMode){ body.response_format={ type:'json_object' }; } }else if(type==='anthropic'){ headers['x-api-key']=p.apiKey; headers['anthropic-version']='2023-06-01'; headers['anthropic-dangerous-direct-browser-access']='true'; body={ model:p.model, system:system.slice(0,16000), max_tokens:maxTokens, messages:messages.map(m=>{ const content=[ { type:'text', text:String(m.text||'').slice(0,14000) } ]; for(const img of (m.images||[]).slice(0,2)){ content.push({ type:'image', source:{ type:'base64', media_type:String(img.mimeType||'image/jpeg'), data:String(img.base64||'') } }); } return { role:m.role==='assistant'?'assistant':'user', content }; }) }; } /* * 一時的なAPI障害に対応。 * 503 / 502 / 500 / 504 / 429 は最大3回まで再試行する。 */ const retryableStatuses=[429,500,502,503,504]; let res=null; let data={}; let rawText=''; for(let attempt=0;attempt<3;attempt++){ try{ res=await fetchWithTimeout( url, { method:'POST', headers, body:JSON.stringify(body) } ); }catch(e){ /* * ネットワーク切断・タイムアウトも、 * まだ再試行回数が残っていれば再試行。 */ if(attempt<2){ await sleep(1000*(attempt+1)); continue; } throw e; } rawText=await res.text(); try{ data=rawText?JSON.parse(rawText):{}; }catch(_){ data={}; } if(res.ok){ break; } const status=Number(res.status||0); if(retryableStatuses.includes(status)&&attempt<2){ const retryAfterRaw=res.headers.get('Retry-After'); const retryAfterSeconds=Number(retryAfterRaw); let waitMs=1000*(attempt+1); if(Number.isFinite(retryAfterSeconds)&&retryAfterSeconds>0){ waitMs=Math.min(retryAfterSeconds*1000,8000); } console.warn( '[AI retry]', `provider=${p.label||p.id}`, `model=${p.model}`, `status=${status}`, `attempt=${attempt+1}/3`, `wait=${waitMs}ms` ); await sleep(waitMs); continue; } const detail= data?.error?.message || data?.message || (rawText?String(rawText).slice(0,300):''); const err=new Error( `HTTP ${status}${detail?`: ${detail}`:''}` ); err.status=status; err.provider=p.label||p.id; err.model=p.model; throw err; } if(!res){ throw new Error('AIへの通信に失敗しました。'); } if(!res.ok){ const detail= data?.error?.message || data?.message || ''; const err=new Error( `HTTP ${res.status}${detail?`: ${detail}`:''}` ); err.status=res.status; err.provider=p.label||p.id; err.model=p.model; throw err; } let out=''; if(type==='gemini'){ out= data?.candidates?.[0]?.content?.parts ?.map(x=>x.text||'') .join('') || ''; }else if(type==='anthropic'){ out= data?.content ?.map(x=>x.text||'') .join('') || ''; }else{ out= typeof data?.choices?.[0]?.message?.content==='string' ?data.choices[0].message.content :''; } if(!out){ throw new Error('AIから回答が返ってきませんでした。'); } return{ text:out, provider:p.label||p.id, model:p.model }; }
